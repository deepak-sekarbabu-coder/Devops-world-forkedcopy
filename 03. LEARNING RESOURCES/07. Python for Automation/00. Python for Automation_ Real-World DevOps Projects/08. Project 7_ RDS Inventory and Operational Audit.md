Think of RDS in two layers.

## AWS control plane

```text
What DB instances exist?

Are they encrypted?
```

Applications connect to the database endpoint using database protocols.

For example:

```text

    |
    | PostgreSQL
    v

```

Project 7 primarily operates on the control plane.

# 11. Enumerate DB Instances

Use:

```python
describe_db_instances()
```

Do not assume one API response contains every database.

Use pagination.

```python
def list_db_instances(
    rds_client
):
    paginator = (
        rds_client.get_paginator(
            "describe_db_instances"
        )
    )

    instances = []

    for page in paginator.paginate():
        instances.extend(
            page.get(
                "DBInstances",
                []
            )
        )

    return instances
```

---

# 12. Why Pagination Matters

A development account might contain:

```text
3 databases
```

A large organization may contain:

```text
hundreds of databases
```

A script that works only because the current account is small is not production-ready.

Use pagination from the beginning.

# 60. Normalize DB Subnet Group

```python
def normalize_subnet_group(
    subnet_group
):
    if not subnet_group:
        return {}

    return {
        "name":
            subnet_group.get(
                "DBSubnetGroupName"
            ),
        "description":
            subnet_group.get(
                "DBSubnetGroupDescription"
            ),
        "vpc_id":
            subnet_group.get(
                "VpcId"
            ),
        "status":
            subnet_group.get(
                "SubnetGroupStatus"
            ),
        "subnets": [
            {
                "subnet_id":
                    subnet.get(
                        "SubnetIdentifier"
                    ),
                "az":
                    (
                        subnet.get(
                            "SubnetAvailabilityZone",
                            {}
                        )
                        .get("Name")
                    ),
                "status":
                    subnet.get(
                        "SubnetStatus"
                    )
            }
            for subnet in subnet_group.get(
                "Subnets",
                []
            )
```

```python
    region,
    tags
):
    members = (
        normalize_cluster_members(
            cluster.get(
                "DBClusterMembers",
                []
            )
        )
    )

    return {
        "resource_type":
            "rds_db_cluster",

        "region":
            region,

        "cluster_identifier":
            cluster.get(
                "DBClusterIdentifier"
            ),

        "cluster_arn":
            cluster.get(
                "DBClusterArn"
            ),

        "engine":
            cluster.get(
                "Engine"
            ),

        "engine_version":
            cluster.get(
                "EngineVersion"
            ),

        "status":
            cluster.get(
                "Status"
            ),

        "endpoint":
            cluster.get(
                "Endpoint"
            ),

        "reader_endpoint":
            cluster.get(
                "ReaderEndpoint"
            ),

        "port":
            cluster.get(
                "Port"
            ),

        "backup_retention_days":
            cluster.get(
                "BackupRetentionPeriod"
            ),

        "backup_window":
            cluster.get(
                "PreferredBackupWindow"
            ),

        "maintenance_window":
            cluster.get(
                "PreferredMaintenanceWindow"
            ),

        "storage_encrypted":
            cluster.get(
                "StorageEncrypted"
            ),

        "kms_key_id":
            cluster.get(
                "KmsKeyId"
            ),

        "deletion_protection":
            cluster.get(
                "DeletionProtection"
            ),

        "security_groups":
            normalize_security_groups(
                cluster.get(
                    "VpcSecurityGroups",
                    []
                )
            ),

        "subnet_group_name":
            cluster.get(
                "DBSubnetGroup"
            ),

        "members":
            members,

        "writer_count":
            sum(
                1
                for member in members
                if member.get(
                    "is_writer"
                )
            ),

        "reader_count":
            sum(
                1
                for member in members
                if not member.get(
                    "is_writer"
                )
            ),

        "tags":
            tags
    }
```

# 244. Region Collector

Conceptually:

```python
def collect_region(
    session,
    region,
    policy
):
    rds = session.client(
        "rds",
        region_name=region
    )

    raw_instances = (
        list_db_instances(
            rds
        )
    )

    raw_clusters = (
        list_db_clusters(
            rds
        )
    )

    instances = []
    clusters = []
    findings = []

    for raw in raw_instances:
        arn = raw.get(
            "DBInstanceArn"
        )

        tags = (
            get_resource_tags(
                rds,
                arn
            )
            if arn
            else {}
        )

        database = (
            normalize_db_instance(
                raw,
                region,
                tags
            )
        )

        instances.append(
            database
        )

        findings.extend(
            evaluate_database(
                database,
                policy
            )
        )

    for raw in raw_clusters:
        arn = raw.get(
            "DBClusterArn"
        )

        tags = (
            get_resource_tags(
                rds,
                arn
            )
            if arn
            else {}
        )

        cluster = (
            normalize_db_cluster(
                raw,
                region,
                tags
            )
        )

        clusters.append(
            cluster
        )

    return {
        "region":
            region,
        "status":
            "SUCCESS",
        "db_instances":
            instances,
        "db_clusters":
            clusters,
        "findings":
            findings
    }
```

In production, handle tag failures separately so one failed tag request does not unnecessarily discard the entire database inventory.
