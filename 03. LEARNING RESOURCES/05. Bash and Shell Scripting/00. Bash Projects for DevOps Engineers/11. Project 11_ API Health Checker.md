# Project 11: API Health Checker

A website can return:

```text
200 OK
```

and still have an unhealthy API.

An API health check should validate more than HTTP connectivity.

It should ask:

```text
Did the request succeed?

Did the API return valid JSON?

Does the JSON contain the required fields?

Does the application itself report healthy?

Was the response fast enough?
```

In this project, we will build:

```text
api-check.sh
```

The script will test API endpoints such as:

```text
GET /health
GET /api/status
```

and validate responses such as:

```json
{
  "status": "healthy"
}
```

A successful check might produce:

```text
==========================================================================
                           API HEALTH REPORT
==========================================================================

URL:                  https://api.example.com/health
Method:               GET
HTTP Status:          200
Response Time:        0.18s
Content Type:         application/json

JSON Validation:      PASSED
Required Field:       status
Expected Value:       healthy
Actual Value:         healthy

API Status:           HEALTHY

==========================================================================
```

A failed application could return:

```json
{
  "status": "unhealthy"
}
```

even with:

```text
HTTP 200
```

Our checker must detect that.

The most important lesson is:

```text
HTTP success and application health are not the same thing.
```

Imagine VERIQTA operates an API:

```text

```

It exposes:

```text

```

# 4. REST API Basics

REST APIs commonly expose endpoints such as:

```text
GET /health
GET /api/status
GET /users
POST /orders
DELETE /sessions/123
```

For health monitoring, we want operations that do not change state.

Prefer:

```text
GET
```

or sometimes:

```text
HEAD
```

for monitoring.

Do not create or delete production resources just to check whether an API is alive.

# 35. Complete Working Script

```bash
#!/usr/bin/env bash

set -u
set -o pipefail

umask 077

# ------------------------------------------------------------------
# API Health Checker
# ------------------------------------------------------------------
#
# Required usage:
#   ./api-check.sh URL EXPECTED_STATUS
#
# Example:
#   ./api-check.sh \
#       https://api.example.com/health \
#       healthy
#
# Exit codes:
#   0 = HEALTHY
#   1 = WARNING
#   2 = CRITICAL
#   3 = Input/configuration error
#
# ------------------------------------------------------------------

URL=""
EXPECTED_STATUS=""

STATUS_FIELD="status"
EXPECTED_HTTP_CODE="200"

CONNECT_TIMEOUT=5
MAX_TIME=10

WARNING_TIME="1.0"
CRITICAL_TIME="3.0"

AUTH_ENV_VAR=""
API_KEY_ENV_VAR=""
API_KEY_HEADER="X-API-Key"

HTTP_CODE="000"
RESPONSE_TIME="0"
CONTENT_TYPE=""

ACTUAL_STATUS=""
JSON_STATUS="NOT CHECKED"
FIELD_STATUS="NOT CHECKED"

CURL_STATUS=0
CURL_ERROR=""

HEALTH_STATUS="UNKNOWN"
HEALTH_REASON=""

RESPONSE_FILE=""
ERROR_FILE=""

SCRIPT_DIR=$(
    cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &&
    pwd
)

readonly SCRIPT_DIR

LOG_DIR="$SCRIPT_DIR/logs"
LOG_FILE="$LOG_DIR/api-health.log"


show_help() {

    cat <<EOF
Usage:
  $0 [options] URL EXPECTED_STATUS

Required positional arguments:
  URL                    API endpoint
  EXPECTED_STATUS        Expected value of the JSON status field

Options:
  --field NAME           JSON field to validate
                         Default: status

  --expect-http CODE     Expected HTTP status
                         Default: 200

  --connect-timeout N    Connection timeout in seconds
                         Default: 5

  --max-time N           Maximum total request time
                         Default: 10

  --warning-time N       Warning latency threshold
                         Default: 1.0

  --critical-time N      Critical latency threshold
                         Default: 3.0

  --bearer-env NAME      Read Bearer token from environment
                         variable NAME

  --api-key-env NAME     Read API key from environment
                         variable NAME

  --api-key-header NAME  API key header
                         Default: X-API-Key

  --help                 Display this help

Examples:
  $0 \
    https://api.example.com/health \
    healthy

  $0 \
    --field state \
    https://api.example.com/status \
    ready

  $0 \
    --expect-http 200 \
    --warning-time 0.5 \
    --critical-time 2.0 \
    https://api.example.com/health \
    healthy

  API_TOKEN='...' \
  $0 \
    --bearer-env API_TOKEN \
    https://api.example.com/health \
    healthy
EOF
}


is_integer() {
    [[ "$1" =~ ^[0-9]+$ ]]
}


is_number() {
    [[ "$1" =~ ^[0-9]+([.][0-9]+)?$ ]]
}


cleanup() {

    if [[ -n "${RESPONSE_FILE:-}" &&
          -f "$RESPONSE_FILE" ]]; then

        rm -f -- "$RESPONSE_FILE"
    fi

    if [[ -n "${ERROR_FILE:-}" &&
          -f "$ERROR_FILE" ]]; then

        rm -f -- "$ERROR_FILE"
    fi
}


trap cleanup EXIT


parse_arguments() {

    local -a positional=()

    while [[ $# -gt 0 ]]; do

        case "$1" in

            --field)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --field requires a value.\n' >&2
                    exit 3
                }

                STATUS_FIELD="$2"
                shift 2
                ;;

            --expect-http)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --expect-http requires a value.\n' >&2
                    exit 3
                }

                if [[ ! "$2" =~ ^[0-9]{3}$ ]]; then

                    printf 'ERROR: Expected HTTP status must be three digits.\n' \
                        >&2

                    exit 3
                fi

                EXPECTED_HTTP_CODE="$2"
                shift 2
                ;;

            --connect-timeout)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --connect-timeout requires a value.\n' >&2
                    exit 3
                }

                if ! is_integer "$2" || (( $2 < 1 )); then

                    printf 'ERROR: Invalid connection timeout.\n' >&2
                    exit 3
                fi

                CONNECT_TIMEOUT="$2"
                shift 2
                ;;

            --max-time)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --max-time requires a value.\n' >&2
                    exit 3
                }

                if ! is_integer "$2" || (( $2 < 1 )); then

                    printf 'ERROR: Invalid maximum request time.\n' >&2
                    exit 3
                fi

                MAX_TIME="$2"
                shift 2
                ;;

            --warning-time)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --warning-time requires a value.\n' >&2
                    exit 3
                }

                if ! is_number "$2"; then

                    printf 'ERROR: Invalid warning threshold.\n' >&2
                    exit 3
                fi

                WARNING_TIME="$2"
                shift 2
                ;;

            --critical-time)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --critical-time requires a value.\n' >&2
                    exit 3
                }

                if ! is_number "$2"; then

                    printf 'ERROR: Invalid critical threshold.\n' >&2
                    exit 3
                fi

                CRITICAL_TIME="$2"
                shift 2
                ;;

            --bearer-env)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --bearer-env requires a variable name.\n' >&2
                    exit 3
                }

                AUTH_ENV_VAR="$2"
                shift 2
                ;;

            --api-key-env)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --api-key-env requires a variable name.\n' >&2
                    exit 3
                }

                API_KEY_ENV_VAR="$2"
                shift 2
                ;;

            --api-key-header)

                [[ $# -ge 2 ]] || {
                    printf 'ERROR: --api-key-header requires a value.\n' >&2
                    exit 3
                }

                API_KEY_HEADER="$2"
                shift 2
                ;;

            --help|-h)

                show_help
                exit 0
                ;;

            --*)

                printf 'ERROR: Unknown option: %s\n' "$1" >&2
                exit 3
                ;;

            *)

                positional+=("$1")
                shift
                ;;
        esac
    done

    if (( ${#positional[@]} != 2 )); then

        printf 'ERROR: URL and EXPECTED_STATUS are required.\n\n' >&2
        show_help
        exit 3
    fi

    URL="${positional[0]}"
    EXPECTED_STATUS="${positional[1]}"
}


check_dependencies() {

    local commands=(
        curl
        jq
        awk
        date
        mkdir
        mktemp
        rm
    )

    local cmd

    for cmd in "${commands[@]}"; do

        if ! command -v "$cmd" >/dev/null 2>&1; then

            printf 'ERROR: Required command missing: %s\n' \
                "$cmd" >&2

            exit 3
        fi
    done
}


validate_inputs() {

    if [[ ! "$URL" =~ ^https?:// ]]; then

        printf 'ERROR: URL must begin with http:// or https://\n' >&2
        exit 3
    fi

    if [[ -z "$EXPECTED_STATUS" ]]; then

        printf 'ERROR: Expected status cannot be empty.\n' >&2
        exit 3
    fi

    if [[ ! "$STATUS_FIELD" =~ ^[A-Za-z_][A-Za-z0-9_-]*$ ]]; then

        printf 'ERROR: Invalid JSON field name: %s\n' \
            "$STATUS_FIELD" >&2

        exit 3
    fi

    if awk \
        -v warning="$WARNING_TIME" \
        -v critical="$CRITICAL_TIME" \
        'BEGIN { exit !(warning >= critical) }'; then

        printf 'ERROR: Warning threshold must be lower than critical threshold.\n' \
            >&2

        exit 3
    fi

    if [[ -n "$AUTH_ENV_VAR" &&
          -n "$API_KEY_ENV_VAR" ]]; then

        printf 'ERROR: Configure one authentication method at a time.\n' \
            >&2

        exit 3
    fi

    if [[ -n "$AUTH_ENV_VAR" &&
          -z "${!AUTH_ENV_VAR:-}" ]]; then

        printf 'ERROR: Authentication environment variable is empty or unset: %s\n' \
            "$AUTH_ENV_VAR" >&2

        exit 3
    fi

    if [[ -n "$API_KEY_ENV_VAR" &&
          -z "${!API_KEY_ENV_VAR:-}" ]]; then

        printf 'ERROR: API-key environment variable is empty or unset: %s\n' \
            "$API_KEY_ENV_VAR" >&2

        exit 3
    fi
}


log_message() {

    local level="$1"
    local message="$2"

    mkdir -p "$LOG_DIR"

    printf '[%s] [%s] %s\n' \
        "$(date '+%Y-%m-%d %H:%M:%S')" \
        "$level" \
        "$message" \
        >> "$LOG_FILE"
}


perform_request() {

    local result
    local -a curl_options

    RESPONSE_FILE=$(mktemp)
    ERROR_FILE=$(mktemp)

    curl_options=(
        -sS
        -o "$RESPONSE_FILE"
        --connect-timeout "$CONNECT_TIMEOUT"
        --max-time "$MAX_TIME"
        -H "Accept: application/json"
        -A "VERIQTA-API-Health-Checker/1.0"
    )

    if [[ -n "$AUTH_ENV_VAR" ]]; then

        curl_options+=(
            -H "Authorization: Bearer ${!AUTH_ENV_VAR}"
        )
    fi

    if [[ -n "$API_KEY_ENV_VAR" ]]; then

        curl_options+=(
            -H "$API_KEY_HEADER: ${!API_KEY_ENV_VAR}"
        )
    fi

    result=$(
        curl \
            "${curl_options[@]}" \
            -w '%{http_code}|%{time_total}|%{content_type}' \
            "$URL" \
            2>"$ERROR_FILE"
    )

    CURL_STATUS=$?

    CURL_ERROR=$(
        cat "$ERROR_FILE"
    )

    IFS='|' read -r \
        HTTP_CODE \
        RESPONSE_TIME \
        CONTENT_TYPE \
        <<< "$result"

    HTTP_CODE="${HTTP_CODE:-000}"
    RESPONSE_TIME="${RESPONSE_TIME:-0}"
    CONTENT_TYPE="${CONTENT_TYPE:-unknown}"
}


classify_transport_failure() {

    case "$CURL_STATUS" in

        5|6)
            HEALTH_STATUS="CRITICAL"
            HEALTH_REASON="DNS resolution failed"
            ;;

        7)
            HEALTH_STATUS="CRITICAL"
            HEALTH_REASON="Connection failed"
            ;;

        28)
            HEALTH_STATUS="CRITICAL"
            HEALTH_REASON="Request timed out"
            ;;

        35)
            HEALTH_STATUS="CRITICAL"
            HEALTH_REASON="TLS/SSL connection failed"
            ;;

        60)
            HEALTH_STATUS="CRITICAL"
            HEALTH_REASON="TLS certificate verification failed"
            ;;

        *)
            HEALTH_STATUS="CRITICAL"

            if [[ -n "$CURL_ERROR" ]]; then
                HEALTH_REASON="$CURL_ERROR"
            else
                HEALTH_REASON="curl failed with exit code $CURL_STATUS"
            fi
            ;;
    esac
}


validate_http_status() {

    if (( CURL_STATUS != 0 )); then

        classify_transport_failure
        return 1
    fi

    if [[ "$HTTP_CODE" != "$EXPECTED_HTTP_CODE" ]]; then

        HEALTH_STATUS="CRITICAL"

        HEALTH_REASON="Expected HTTP $EXPECTED_HTTP_CODE but received HTTP $HTTP_CODE"

        return 1
    fi

    return 0
}


validate_content_type() {

    if [[ "$CONTENT_TYPE" == application/json* ||
          "$CONTENT_TYPE" == *"+json"* ]]; then

        return 0
    fi

    HEALTH_STATUS="WARNING"
    HEALTH_REASON="Unexpected Content-Type: $CONTENT_TYPE"

    return 1
}


validate_json() {

    if jq empty "$RESPONSE_FILE" >/dev/null 2>&1; then

        JSON_STATUS="PASSED"
        return 0
    fi

    JSON_STATUS="FAILED"

    HEALTH_STATUS="CRITICAL"
    HEALTH_REASON="Response body is not valid JSON"

    return 1
}


validate_required_field() {

    local jq_filter

    jq_filter=".${STATUS_FIELD}"

    if ! jq -e \
        "$jq_filter != null" \
        "$RESPONSE_FILE" \
        >/dev/null 2>&1; then

        FIELD_STATUS="FAILED"

        HEALTH_STATUS="CRITICAL"
        HEALTH_REASON="Required JSON field is missing: $STATUS_FIELD"

        return 1
    fi

    ACTUAL_STATUS=$(
        jq -r \
            "$jq_filter" \
            "$RESPONSE_FILE"
    )

    FIELD_STATUS="PASSED"

    return 0
}


compare_expected_status() {

    if [[ "$ACTUAL_STATUS" == "$EXPECTED_STATUS" ]]; then

        HEALTH_STATUS="HEALTHY"
        HEALTH_REASON="API returned expected application status"

        return 0
    fi

    HEALTH_STATUS="CRITICAL"

    HEALTH_REASON="Expected $STATUS_FIELD=$EXPECTED_STATUS but received $STATUS_FIELD=$ACTUAL_STATUS"

    return 1
}


response_time_at_least() {

    local threshold="$1"

    awk \
        -v value="$RESPONSE_TIME" \
        -v threshold="$threshold" \
        'BEGIN {
            exit !(value >= threshold)
        }'
}


evaluate_response_time() {

    if [[ "$HEALTH_STATUS" == "CRITICAL" ]]; then
        return
    fi

    if response_time_at_least "$CRITICAL_TIME"; then

        HEALTH_STATUS="CRITICAL"

        HEALTH_REASON="API response time exceeded critical threshold"

        return
    fi

    if response_time_at_least "$WARNING_TIME"; then

        HEALTH_STATUS="WARNING"

        HEALTH_REASON="API response time exceeded warning threshold"
    fi
}


print_report() {

    printf '%s\n' \
        "=========================================================================="

    printf '%s\n' \
        "                           API HEALTH REPORT"

    printf '%s\n' \
        "=========================================================================="

    printf '\n%-24s %s\n' \
        "URL:" "$URL"

    printf '%-24s %s\n' \
        "Method:" "GET"

    printf '%-24s %s\n' \
        "HTTP Status:" "$HTTP_CODE"

    printf '%-24s %ss\n' \
        "Response Time:" "$RESPONSE_TIME"

    printf '%-24s %s\n' \
        "Content Type:" "$CONTENT_TYPE"

    printf '\n%s\n' \
        "----------------------------- VALIDATION ---------------------------------"

    printf '%-24s %s\n' \
        "Expected HTTP:" "$EXPECTED_HTTP_CODE"

    printf '%-24s %s\n' \
        "JSON Validation:" "$JSON_STATUS"

    printf '%-24s %s\n' \
        "Required Field:" "$STATUS_FIELD"

    printf '%-24s %s\n' \
        "Field Validation:" "$FIELD_STATUS"

    printf '%-24s %s\n' \
        "Expected Value:" "$EXPECTED_STATUS"

    if [[ -n "$ACTUAL_STATUS" ]]; then

        printf '%-24s %s\n' \
            "Actual Value:" "$ACTUAL_STATUS"
    else

        printf '%-24s %s\n' \
            "Actual Value:" "N/A"
    fi

    printf '\n%s\n' \
        "--------------------------------------------------------------------------"

    printf '%-24s %s\n' \
        "API Status:" "$HEALTH_STATUS"

    printf '\nReason:\n%s\n' \
        "$HEALTH_REASON"

    printf '%s\n' \
        "=========================================================================="
}


write_log() {

    local auth_mode="none"

    if [[ -n "$AUTH_ENV_VAR" ]]; then
        auth_mode="bearer"
    elif [[ -n "$API_KEY_ENV_VAR" ]]; then
        auth_mode="api_key"
    fi

    log_message \
        "INFO" \
        "url=$URL http=$HTTP_CODE expected_http=$EXPECTED_HTTP_CODE response_time=$RESPONSE_TIME json=$JSON_STATUS field=$STATUS_FIELD expected_value=$EXPECTED_STATUS actual_value=${ACTUAL_STATUS:-NA} auth=$auth_mode health=$HEALTH_STATUS"
}


return_status() {

    case "$HEALTH_STATUS" in

        HEALTHY)
            return 0
            ;;

        WARNING)
            return 1
            ;;

        CRITICAL)
            return 2
            ;;

        *)
            return 3
            ;;
    esac
}


main() {

    parse_arguments "$@"

    check_dependencies

    validate_inputs

    perform_request

    if validate_http_status; then

        validate_content_type || true

        if validate_json; then

            if validate_required_field; then

                compare_expected_status || true
            fi
        fi
    fi

    evaluate_response_time

    print_report

    write_log

    return_status
}


main "$@"
exit $?
```
